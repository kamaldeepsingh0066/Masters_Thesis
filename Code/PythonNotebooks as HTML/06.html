# MAPE FUNCTION
# ==========================================================

def calculate_mape(
    y_true,
    y_pred
):

    y_true = np.where(
        y_true == 0,
        1e-8,
        y_true
    )

    mape = np.mean(
        np.abs(
            (y_true - y_pred)
            / y_true
        )
    ) * 100

    return mape

# ==========================================================
# 6. EVALUATION FUNCTIONS
# ==========================================================

def evaluate_model(
    y_true,
    y_pred
):

    mae = mean_absolute_error(
        y_true,
        y_pred
    )

    rmse = np.sqrt(
        mean_squared_error(
            y_true,
            y_pred
        )
    )

    mape = calculate_mape(
        y_true,
        y_pred
    )

    r2 = r2_score(
        y_true,
        y_pred
    )

    return (
        mae,
        rmse,
        mape,
        r2
    )

def get_period_split(df, period_name):

    period = PERIODS[period_name]

    # Filter period
    df_period = df[
        df["datetime"] >= period["start"]
    ].copy()

    # Train
    train_df = df_period[
        df_period["datetime"] <= period["train_end"]
    ].copy()

    # Validation
    valid_df = df_period[
        (df_period["datetime"] > period["train_end"]) &
        (df_period["datetime"] <= period["valid_end"])
    ].copy()

    # Test
    test_df = df_period[
        df_period["datetime"] > period["valid_end"]
    ].copy()

    return train_df, valid_df, test_df

####################################################################

def get_model_data(
    train_df,
    valid_df,
    test_df,
    target_name,
    horizon
):

    features = FEATURES[target_name]

    target_col = f"target_h{horizon}"

    X_train = train_df[features]
    y_train = train_df[target_col]

    X_valid = valid_df[features]
    y_valid = valid_df[target_col]

    X_test = test_df[features]
    y_test = test_df[target_col]

    return (
        X_train,
        y_train,
        X_valid,
        y_valid,
        X_test,
        y_test
    )

def run_persistence(
    valid_df,
    target_name,
    horizon
):

    target_col = f"target_h{horizon}"

    base_target = TARGETS[target_name]

    lag_col = f"{base_target}_lag_{horizon}"

    y_true = valid_df[target_col]

    y_pred = valid_df[lag_col]

    mae, rmse, mape, r2 = evaluate_model(
        y_true,
        y_pred
    )

    return mae, rmse, mape, r2

#################################################################

def run_random_forest(
    X_train,
    y_train,
    X_valid,
    y_valid
):

    model = RandomForestRegressor(
        n_estimators=200,
        max_depth=15,
        min_samples_leaf=5,
        random_state=42,
        n_jobs=-1
    )

    model.fit(
        X_train,
        y_train
    )

    y_pred = model.predict(
        X_valid
    )

    mae, rmse, mape, r2 = evaluate_model(
        y_valid,
        y_pred
    )

    return model, mae, rmse, mape, r2

def run_model(model_name, 
              X_train=None, y_train=None, 
              X_valid=None, y_valid=None, 
              valid_df=None, target_name=None, horizon=None ):

    # =====================================
    # Persistence
    # =====================================
    if model_name == "Persistence":

        mae, rmse, mape, r2 = run_persistence(
            valid_df=valid_df,
            target_name=target_name,
            horizon=horizon
        )

        return {
            "model": None,
            "mae": mae,
            "rmse": rmse,
            "mape": mape,
            "r2": r2
        }

    # =====================================
    # Random Forest
    # =====================================
    elif model_name == "RandomForest":

        model, mae, rmse, mape, r2 = run_random_forest(
            X_train,
            y_train,
            X_valid,
            y_valid
        )

        return {
            "model": model,
            "mae": mae,
            "rmse": rmse,
            "mape": mape,
            "r2": r2
        }

    # =====================================
    # XGBoost
    # =====================================
    elif model_name == "XGBoost":

        model, mae, rmse, mape, r2 = run_xgboost(
            X_train,
            y_train,
            X_valid,
            y_valid
        )

        return {
            "model": model,
            "mae": mae,
            "rmse": rmse,
            "mape": mape,
            "r2": r2
        }

    # =====================================
    # LightGBM
    # =====================================
    elif model_name == "LightGBM":

        model, mae, rmse, mape, r2 = run_lightgbm(
            X_train,
            y_train,
            X_valid,
            y_valid
        )

        return {
            "model": model,
            "mae": mae,
            "rmse": rmse,
            "mape": mape,
            "r2": r2
        }

    # =====================================
    # LSTM
    # =====================================
    elif model_name == "LSTM":

        model, mae, rmse, mape, r2 = run_lstm(
            X_train,
            y_train,
            X_valid,
            y_valid
        )

        return {
            "model": model,
            "mae": mae,
            "rmse": rmse,
            "mape": mape,
            "r2": r2
        }

    # =====================================
    # Unknown Model
    # =====================================
    else:

        raise ValueError(
            f"Unknown model: {model_name}"
        )

# 10. EXPERIMENT RUNNER
# ==========================================================
def run_experiment(
    dataset,
    period_name,
    target_name,
    horizon,
    model_name,
    results
):

    # Split data
    train_df, valid_df, test_df = get_period_split(
        dataset,
        period_name
    )

    # Features + Target
    (
        X_train,
        y_train,
        X_valid,
        y_valid,
        X_test,
        y_test
    ) = get_model_data(
        train_df,
        valid_df,
        test_df,
        target_name,
        horizon
    )

    # Run model
    result = run_model(
        model_name=model_name,

        X_train=X_train,
        y_train=y_train,

        X_valid=X_valid,
        y_valid=y_valid,

        valid_df=valid_df,

        target_name=target_name,
        horizon=horizon
    )

    # Save result
    results = add_result(
        results,
        period_name,
        target_name,
        horizon,
        model_name,
        result["mae"],
        result["rmse"],
        result["mape"],
        result["r2"]
    )

    return results
